cmake_minimum_required(VERSION 3.16)
project(harmony_agent LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(VENDOR "${ROOT}/vendor")
set(HDC "${VENDOR}/developtools_hdc-OpenHarmony-6.0-Release")
set(SSL "${VENDOR}/openssl-openssl-3.0.16")
if(NOT EXISTS "${SSL}/libssl.a")
  message(FATAL_ERROR "Run scripts/prepare-native.sh before building")
endif()
file(GLOB SECUREC "${VENDOR}/third_party_bounds_checking_function-master/src/*.c")
add_library(securec STATIC ${SECUREC})
target_include_directories(securec PUBLIC "${VENDOR}/third_party_bounds_checking_function-master/include")
add_library(lz4 STATIC "${VENDOR}/lz4-1.10.0/lib/lz4.c" "${VENDOR}/lz4-1.10.0/lib/lz4hc.c" "${VENDOR}/lz4-1.10.0/lib/lz4frame.c" "${VENDOR}/lz4-1.10.0/lib/xxhash.c")
target_include_directories(lz4 PUBLIC "${VENDOR}/lz4-1.10.0/lib")
file(GLOB HDC_COMMON "${HDC}/src/common/*.cpp")
file(GLOB HDC_HOST "${HDC}/src/host/*.cpp")
list(FILTER HDC_COMMON EXCLUDE REGEX "/(hdc_huks|password|uart)\\.cpp$")
list(FILTER HDC_HOST EXCLUDE REGEX "/(main|client|ext_client|host_usb|host_uart)\\.cpp$")
file(WRITE "${CMAKE_BINARY_DIR}/hdc_hash_gen.h" "#pragma once\n#define HDC_MSG_HASH \"\"\n")
add_library(agent_hdc SHARED bridge.cpp ${HDC_COMMON} ${HDC_HOST})
target_include_directories(agent_hdc PRIVATE "${HDC}/src/common" "${HDC}/src/host" "${SSL}/include" "${CMAKE_BINARY_DIR}")
target_compile_definitions(agent_hdc PRIVATE HDC_HOST HOST_OHOS HARMONY_PROJECT HDC_NO_USB HDC_SUPPORT_ENCRYPT_TCP)
target_compile_options(agent_hdc PRIVATE -Wno-deprecated-declarations)
target_link_libraries(agent_hdc PRIVATE "${SSL}/libssl.a" "${SSL}/libcrypto.a" lz4 securec uv ace_napi.z hilog_ndk.z)
